function calcularDoseEritropoetina(config) { const { pesoKg, idade, indicacao, via = 'subcutânea', ferroAdequado = false, esquema = 'inicial' } = config; let dose, frequencia, observacoes; // Baseado nas bulas de eritropoetina (alfa, beta, darbepoetina) if (indicacao === 'anemia renal crônica') { if (via === 'subcutânea') { dose = esquema === 'inicial' ? '50-100 U/kg' : '25-50 U/kg'; frequencia = '3 vezes por semana'; } else { // intravenosa dose = esquema === 'inicial' ? '25-100 U/kg' : '12.5-25 U/kg'; frequencia = '3 vezes por semana'; } observacoes = 'Ajustar para manter Hb 10-12 g/dL. Reduzir dose se Hb > 12 g/dL.'; } else if (indicacao === 'anemia crítica (UTI)') { dose = '40.000 U'; frequencia = '1 vez por semana (esquema fixo)'; observacoes = 'Esquema fixo, não baseado em peso. Avaliar após 4 semanas.'; } else if (indicacao === 'quimioterapia') { dose = '150 U/kg'; frequencia = '3 vezes por semana (SC) ou 40.000 U semanal'; observacoes = 'Iniciar se Hb < 10 g/dL. Suspender se Hb > 12 g/dL.'; } // Cálculo base para renal (exemplo) let doseCalculada = null; if (indicacao === 'anemia renal crônica' && pesoKg) { const doseMin = via === 'subcutânea' ? 50 : 25; const doseMax = via === 'subcutânea' ? 100 : 100; doseCalculada = { minima: (pesoKg * doseMin).toFixed(0), maxima: (pesoKg * doseMax).toFixed(0) }; } // Contraindicações importantes const contraindicacoes = [ 'Hipertensão não controlada', 'Histórico de trombose', 'Alergia ao produto', 'Policitemia vera' ]; const alertas = [ 'Monitorar pressão arterial semanalmente', 'Ferro sérico e saturação devem ser adequados antes de iniciar', 'Suspender se Hb > 12 g/dL', 'Risco de eventos tromboembólicos', 'Ajustar dose mensalmente conforme resposta' ]; return { indicacao, via, doseRecomendada: dose, frequencia, doseCalculada, observacoes, contraindicacoes, alertas, necessidadeFerro: !ferroAdequado ? 'FERRO INSUFICIENTE: Reposição obrigatória antes de iniciar' : 'Ferro adequado para início' }; }
